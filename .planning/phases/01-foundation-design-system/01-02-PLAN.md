Implement `parseAxiosError` fully. Use `axios.isAxiosError(err)`, switch on `err.response?.status`:
- no response → kind: "network", status: 0
- 400/422 with array detail → kind: "validation", parse fieldErrors from `detail[]` (each has `loc[], msg, type`); concatenate `loc` joined by "." for `field`

**`frontend/src/api/client.ts`**:
```ts
import axios, { type AxiosInstance } from "axios";
import { ApiError, parseAxiosError } from "./errors";

const baseURL = import.meta.env.VITE_API_BASE_URL ?? "http://localhost:8000/api/v1";

export const client: AxiosInstance = axios.create({
  baseURL,
  timeout: 60_000,
  withCredentials: false,
  headers: { "Content-Type": "application/json" }
});

client.interceptors.response.use(
  (res) => res,
  (err) => {
    // Do NOT log request/response bodies (PII).
    throw parseAxiosError(err);
  }
);

export function isApiError(err: unknown): err is ApiError {
  return err instanceof ApiError;
}
```

Create one module per resource group. Each module exports an object with named functions returning typed promises. Pattern:

```ts
// frontend/src/api/endpoints/upload.ts
import { client } from "../client";
import type {
  ResumeResponse,
  ResumeListResponse,
  ResumeBatchParseResponse,
  UploadStatus
} from "../types";

export const uploadApi = {
  async list(params?: {
    upload_status?: UploadStatus;
    uploaded_by_user_id?: string;
    limit?: number;
    offset?: number;
  }): Promise<ResumeListResponse> {
    const { data } = await client.get<ResumeListResponse>("/upload/", { params });
    return data;
  },

  async get(resumeId: string): Promise<ResumeResponse> {
    const { data } = await client.get<ResumeResponse>(`/upload/${resumeId}`);
    return data;
  },

  async update(
    resumeId: string,
    body: Partial<Pick<ResumeResponse, "original_file_name" | "upload_status">>
  ): Promise<ResumeResponse> {
    const { data } = await client.patch<ResumeResponse>(`/upload/${resumeId}`, body);
    return data;
  },

  async remove(resumeId: string, deleteFile = false): Promise<{ deleted: boolean; resume_id: string }> {
    const { data } = await client.delete(`/upload/${resumeId}`, { params: { delete_file: deleteFile } });
    return data;
  },

  async batchParse(files: File[], uploaded_by_user_id?: string): Promise<ResumeBatchParseResponse> {
    const fd = new FormData();
    files.forEach((f) => fd.append("files", f));
    if (uploaded_by_user_id) fd.append("uploaded_by_user_id", uploaded_by_user_id);
    const { data } = await client.post<ResumeBatchParseResponse>("/upload/batch-parse", fd, {
      headers: { "Content-Type": "multipart/form-data" },
      timeout: 300_000 // 5 min for sync LLM batch
    });
    return data;
  }
};
```
