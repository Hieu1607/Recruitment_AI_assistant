**`frontend/index.html`** — note the inline script that resolves the saved theme BEFORE React hydrates (prevents FOUC on dark-mode reload):
```html
<!doctype html>
<html lang="en" class="h-full">
  <head>
    <meta charset="UTF-8" />
    <link rel="icon" type="image/svg+xml" href="/favicon.svg" />
    <meta name="viewport" content="width=device-width, initial-scale=1.0" />
    <meta name="theme-color" content="#FAFAF7" />
    <title>RecruitAI — Editorial Intelligence</title>
    <script>
      // Pre-hydration theme resolver — keep in sync with src/lib/theme.ts STORAGE_KEY.
      // Sets data-theme on <html> before first paint to avoid flash-of-wrong-theme.
      (function () {
        try {
          var saved = localStorage.getItem("recruitai.theme");
          var theme = saved === "light" || saved === "dark" || saved === "system" ? saved : "system";
          var applied = theme === "system"
            ? (window.matchMedia && window.matchMedia("(prefers-color-scheme: dark)").matches ? "dark" : "light")
            : theme;
          document.documentElement.setAttribute("data-theme", applied);
        } catch (e) {
          document.documentElement.setAttribute("data-theme", "light");
        }
      })();
    </script>
  </head>
  <body class="h-full antialiased">
    <div id="root"></div>
    <script type="module" src="/src/main.tsx"></script>
  </body>
</html>
```
